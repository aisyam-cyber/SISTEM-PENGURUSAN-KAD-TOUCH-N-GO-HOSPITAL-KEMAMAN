/**
 * Muat Turun Data Kad dari Backend
 */
async function fetchAllCards() {
    const token = localStorage.getItem("tng_session_token");
    const tbody = document.getElementById("cardTableBody");

    try {
        const response = await fetch(API_URL, {
            method: "POST",
            headers: { "Content-Type": "text/plain" },
            body: JSON.stringify({ action: "GET_ALL_CARDS", token: token })
        });

        const result = await response.json();
        if (result.success) {
            globalCardsData = result.data;
            renderCardRows(globalCardsData);
        } else {
            tbody.innerHTML = `<tr><td colspan="8" style="color:red; text-align:center;">${result.message}</td></tr>`;
        }
    } catch (e) {
        tbody.innerHTML = `<tr><td colspan="8" style="color:red; text-align:center;">Ralat sambungan: ${e.message}</td></tr>`;
    }
}

/**
 * Render Baris Jadual Kad
 */
function renderCardRows(cards) {
    const tbody = document.getElementById("cardTableBody");
    if (!cards || cards.length === 0) {
        tbody.innerHTML = `<tr><td colspan="8" style="text-align:center;">Tiada rekod kad ditemui.</td></tr>`;
        return;
    }

    tbody.innerHTML = cards.map(c => `
        <tr>
            <td><strong>${c.cardId}</strong></td>
            <td>${c.cardNumber}</td>
            <td>${c.cardName}</td>
            <td>${c.cardType}</td>
            <td><span class="badge-status ${c.status === 'AKTIF' || c.status === 'DALAM GUNA' ? 'active' : 'warning'}">${c.status}</span></td>
            <td>${c.vehicleAssigned}</td>
            <td style="font-weight:bold; color:${c.balance < 50 ? 'var(--danger-color)' : 'inherit'};">RM ${c.balance.toFixed(2)}</td>
            <td>
                <button class="btn-icon" title="Edit Kad" onclick="openEditCardModal('${c.cardId}')"><i class="fa-solid fa-pen-to-square"></i></button>
                <button class="btn-icon danger" title="Arkibkan Kad" onclick="confirmArchiveCard('${c.cardId}')"><i class="fa-solid fa-box-archive"></i></button>
            </td>
        </tr>
    `).join('');
}

/**
 * Carian & Penapisan Kad
 */
function filterCards() {
    const query = document.getElementById("cardSearchInput").value.toLowerCase();
    const filtered = globalCardsData.filter(c => 
        c.cardNumber.toLowerCase().includes(query) || 
        c.cardName.toLowerCase().includes(query)
    );
    renderCardRows(filtered);
}

function openCreateCardModal() {
    document.getElementById("cardModalTitle").textContent = "Daftar Kad Touch 'n Go";
    document.getElementById("modalCardId").value = "";
    document.getElementById("modalCardNo").value = "";
    document.getElementById("modalCardNo").disabled = false;
    document.getElementById("modalCardName").value = "";
    document.getElementById("modalCardBalance").value = "100.00";
    document.getElementById("cardModal").classList.remove("hidden");
}

function closeCardModal() {
    document.getElementById("cardModal").classList.add("hidden");
}

async function handleSaveCard(event) {
    event.preventDefault();
    const user = JSON.parse(localStorage.getItem("tng_user") || "{}");
    const token = localStorage.getItem("tng_session_token");
    const cardId = document.getElementById("modalCardId").value;
    const alertBox = document.getElementById("cardFormAlert");

    const payloadData = {
        cardNumber: document.getElementById("modalCardNo").value,
        cardName: document.getElementById("modalCardName").value,
        cardType: document.getElementById("modalCardType").value,
        initialBalance: document.getElementById("modalCardBalance").value,
        storageLocation: document.getElementById("modalStorageLocation").value
    };

    const action = cardId ? "UPDATE_CARD" : "CREATE_CARD";
    if (cardId) payloadData.cardId = cardId;

    try {
        const response = await fetch(API_URL, {
            method: "POST",
            headers: { "Content-Type": "text/plain" },
            body: JSON.stringify({
                action: action,
                token: token,
                currentUserRole: user.role,
                data: payloadData
            })
        });

        const result = await response.json();
        if (result.success) {
            closeCardModal();
            await fetchAllCards();
        } else {
            showAlert(alertBox, result.message, "error");
        }
    } catch (e) {
        showAlert(alertBox, "Ralat menyimpan kad.", "error");
    }
}

async function confirmArchiveCard(cardId) {
    if (confirm(`Adakah anda pasti untuk mengarkibkan Kad ID: ${cardId}?`)) {
        const token = localStorage.getItem("tng_session_token");
        const user = JSON.parse(localStorage.getItem("tng_user") || "{}");

        await fetch(API_URL, {
            method: "POST",
            headers: { "Content-Type": "text/plain" },
            body: JSON.stringify({
                action: "ARCHIVE_CARD",
                token: token,
                currentUserRole: user.role,
                data: { cardId: cardId }
            })
        });
        await fetchAllCards();
    }
}
